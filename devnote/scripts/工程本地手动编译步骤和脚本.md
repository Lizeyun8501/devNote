## 一、前置环境依赖

| 组件 | 版本要求 | 安装/验证命令 |
|------|---------|--------------|
| Flutter SDK | 3.29.x (stable) | `flutter doctor -v` |
| Dart SDK | ^3.7.2（随 Flutter） | `dart --version` |
| Rust toolchain | stable | `rustc --version`, `cargo --version` |
| Go | 1.25+ | `go version` |
| Protocol Buffers | `protoc` 任意版本 | `protoc --version` |
| Visual Studio 2022 | C++ 桌面开发 workload | `flutter doctor` |

> **注意**：`protoc` 是编译 Rust 核心的**必需**依赖（`devnote-grpc/build.rs` 使用 `tonic-build` 编译 `.proto`）。Windows 可通过 [protobuf releases](https://github.com/protocolbuffers/protobuf/releases) 或 `choco install protoc` 安装，并确保加入 `PATH`。

## 三、手动编译步骤

### 1. 编译 Rust FFI 核心库

```powershell
cd e:\01project\DevNote\devNote\devNote\rust-core

# Debug 模式
cargo build -p devnote-ffi

# Release 模式
cargo build -p devnote-ffi --release
```

**产物位置**：
- Debug：`rust-core/target/debug/devnote_ffi.dll`
- Release：`rust-core/target/release/devnote_ffi.dll`

### 2. 将 DLL 复制到 Flutter Windows 输出目录

Flutter Windows 应用运行时需在 `.exe` 同级目录找到该 DLL：

```powershell
# 根据构建模式选择来源
$src = "e:\01project\DevNote\devNote\devNote\rust-core\target\release\devnote_ffi.dll"
# 或 debug: $src = "...\target\debug\devnote_ffi.dll"

# 复制到 flutter build 目录（需先执行过 flutter build windows 才有此目录）
$dst = "e:\01project\DevNote\devNote\devNote\build\windows\x64\runner\Release"
Copy-Item $src $dst -Force

# 如果是 Debug 模式运行
# $dst = "...\build\windows\x64\runner\Debug"
```

### 3. 编译并运行 Flutter 前端

```powershell
cd e:\01project\DevNote\devNote\devNote

flutter pub get

# 编译 Windows Release
flutter build windows --release

# 或开发调试运行
flutter run -d windows
```

### 4. 编译 Go 后端服务

**业务服务器 (`business-server`)**：
```powershell
cd e:\01project\DevNote\devNote\devNote\business-server
go build -o server.exe ./cmd/server
```

**同步服务器 (`sync-server`)**：
```powershell
cd e:\01project\DevNote\devNote\devNote\sync-server
go build -o server.exe ./cmd/server
```

## 四、Windows 一键编译脚本

在项目根目录 `e:\01project\DevNote\devNote\devNote\` 下创建 `build_local.ps1`：

```powershell
<#
.SYNOPSIS
    DevNote 本地手动编译脚本 (Windows)
.DESCRIPTION
    编译 Rust FFI、Flutter Windows、Go 后端服务
.PARAMETER Target
    编译目标: all | rust | flutter | go-business | go-sync
.PARAMETER Mode
    构建模式: debug | release
.EXAMPLE
    .\build_local.ps1 -Target all -Mode release
    .\build_local.ps1 -Target rust -Mode debug
#>
param(
    [ValidateSet("all", "rust", "flutter", "go-business", "go-sync")]
    [string]$Target = "all",

    [ValidateSet("debug", "release")]
    [string]$Mode = "release"
)

$ErrorActionPreference = "Stop"
$ProjectRoot = $PSScriptRoot
$RustCoreDir = Join-Path $ProjectRoot "rust-core"
$FlutterDir = $ProjectRoot
$BusinessServerDir = Join-Path $ProjectRoot "business-server"
$SyncServerDir = Join-Path $ProjectRoot "sync-server"

# ── 前置检查 ──
function Test-Command {
    param([string]$Cmd, [string]$Name)
    if (!(Get-Command $Cmd -ErrorAction SilentlyContinue)) {
        Write-Error "缺少依赖: $Name ($Cmd)。请先安装并加入 PATH。"
    }
    Write-Host "[OK] $Name`: $(& $Cmd --version 2>$null | Select-Object -First 1)" -ForegroundColor Green
}

Write-Host "`n=== DevNote 本地编译 ===`n" -ForegroundColor Cyan
Test-Command "flutter" "Flutter SDK"
Test-Command "dart" "Dart SDK"
Test-Command "cargo" "Rust Cargo"
Test-Command "go" "Go"
Test-Command "protoc" "Protocol Buffers"

$cargoProfile = if ($Mode -eq "release") { "--release" } else { "" }
$flutterMode = if ($Mode -eq "release") { "--release" } else { "--debug" }
$runnerDir = if ($Mode -eq "release") { "Release" } else { "Debug" }

# ── 编译 Rust FFI ──
function Build-Rust {
    Write-Host "`n[1/4] 编译 Rust FFI (devnote-ffi) [$Mode]..." -ForegroundColor Cyan
    Push-Location $RustCoreDir
    try {
        cargo build -p devnote-ffi --target x86_64-pc-windows-msvc $cargoProfile
        if ($LASTEXITCODE -ne 0) { throw "Rust 编译失败" }

        $dllName = "devnote_ffi.dll"
        $src = if ($Mode -eq "release") {
            Join-Path $RustCoreDir "target\x86_64-pc-windows-msvc\release\$dllName"
        } else {
            Join-Path $RustCoreDir "target\x86_64-pc-windows-msvc\debug\$dllName"
        }

        # 如果已存在 Flutter 构建目录，自动复制 DLL
        $flutterBuildDir = Join-Path $FlutterDir "build\windows\x64\runner\$runnerDir"
        if (Test-Path $flutterBuildDir) {
            Copy-Item $src $flutterBuildDir -Force
            Write-Host "[OK] 已复制 DLL 到: $flutterBuildDir" -ForegroundColor Green
        } else {
            Write-Host "[WARN] Flutter 构建目录不存在，跳过自动复制 DLL。请手动复制到 build/windows/x64/runner/$runnerDir/" -ForegroundColor Yellow
        }
    } finally {
        Pop-Location
    }
}

# ── 编译 Flutter ──
function Build-Flutter {
    Write-Host "`n[2/4] 编译 Flutter Windows [$Mode]..." -ForegroundColor Cyan
    Push-Location $FlutterDir
    try {
        flutter pub get
        if ($LASTEXITCODE -ne 0) { throw "flutter pub get 失败" }

        flutter build windows $flutterMode
        if ($LASTEXITCODE -ne 0) { throw "Flutter 编译失败" }

        # 编译完成后再次确保 DLL 到位
        $src = if ($Mode -eq "release") {
            Join-Path $RustCoreDir "target\x86_64-pc-windows-msvc\release\devnote_ffi.dll"
        } else {
            Join-Path $RustCoreDir "target\x86_64-pc-windows-msvc\debug\devnote_ffi.dll"
        }
        $dst = Join-Path $FlutterDir "build\windows\x64\runner\$runnerDir"
        if (Test-Path $src) {
            Copy-Item $src $dst -Force
            Write-Host "[OK] DLL 已确保到位: $dst" -ForegroundColor Green
        }
    } finally {
        Pop-Location
    }
}

# ── 编译 Go Business Server ──
function Build-GoBusiness {
    Write-Host "`n[3/4] 编译 Go Business Server..." -ForegroundColor Cyan
    Push-Location $BusinessServerDir
    try {
        go build -o server.exe ./cmd/server
        if ($LASTEXITCODE -ne 0) { throw "Go business-server 编译失败" }
        Write-Host "[OK] 产物: $BusinessServerDir\server.exe" -ForegroundColor Green
    } finally {
        Pop-Location
    }
}

# ── 编译 Go Sync Server ──
function Build-GoSync {
    Write-Host "`n[4/4] 编译 Go Sync Server..." -ForegroundColor Cyan
    Push-Location $SyncServerDir
    try {
        go build -o server.exe ./cmd/server
        if ($LASTEXITCODE -ne 0) { throw "Go sync-server 编译失败" }
        Write-Host "[OK] 产物: $SyncServerDir\server.exe" -ForegroundColor Green
    } finally {
        Pop-Location
    }
}

# ── 执行 ──
switch ($Target) {
    "rust"        { Build-Rust }
    "flutter"     { Build-Flutter }
    "go-business" { Build-GoBusiness }
    "go-sync"     { Build-GoSync }
    "all" {
        Build-Rust
        Build-Flutter
        Build-GoBusiness
        Build-GoSync
    }
}

Write-Host "`n=== 编译完成 ===" -ForegroundColor Cyan
```

### 使用方法

```powershell
# 一键编译全部（Release）
cd e:\01project\DevNote\devNote\devNote
.\build_local.ps1 -Target all -Mode release

# 仅编译 Rust FFI（Debug）
.\build_local.ps1 -Target rust -Mode debug

# 仅编译 Flutter
.\build_local.ps1 -Target flutter -Mode release

# 仅编译 Go 后端
.\build_local.ps1 -Target go-business
.\build_local.ps1 -Target go-sync
```

## 五、常见问题

1. **`protoc` 未找到**  
   安装 Protocol Buffers 并确保 `protoc.exe` 在 `PATH` 中。Rust 的 `devnote-grpc` crate 编译 `.proto` 文件时必需。

2. **Rust 编译时链接器错误**  
   确保已安装 **Visual Studio 2022 的 C++ 桌面开发** workload。Flutter Windows 和 Rust MSVC target 均依赖此环境。

3. **Flutter 运行时提示找不到 `devnote_ffi.dll`**  
   确认 DLL 已复制到 `build/windows/x64/runner/Debug/`（或 `Release/`）目录，与 `devnote.exe` 同级。

4. **Android 编译**  
   当前 `android/app/src/main/jniLibs` 已包含预编译的 `libdevnote_ffi.so`（arm64-v8a / armeabi-v7a / x86_64），可直接 `flutter run`。如需重新编译 Android 版 Rust 库，需要配置 Android NDK 并使用 `cargo build --target aarch64-linux-android`，具体可参考 [`scripts/build_rust.sh`](file:///e:/01project/DevNote/devNote/devNote/scripts/build_rust.sh)。
